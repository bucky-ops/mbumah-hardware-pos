    58→function getAvatarGradient(name: string): string {
    59→  let hash = 0;
    60→  for (let i = 0; i < name.length; i++) {
    61→    hash = name.charCodeAt(i) + ((hash << 5) - hash);
    62→  }
    63→  return AVATAR_GRADIENTS[Math.abs(hash) % AVATAR_GRADIENTS.length];
    64→}

   357→          });
   358→        }
   359→      }
   360→    });

   385→    });
