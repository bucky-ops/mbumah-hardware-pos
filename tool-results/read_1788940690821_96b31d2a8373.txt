    30→//   Step 1 — Customer & Payment Method

    32→//   Step 3 — Review & Confirm (order summary + Process Payment)
    33→
    34→export interface CheckoutDialogProps {
    35→  open: boolean;
    36→  onOpenChange: (open: boolean) => void;
    37→  finalTotal: number;
    38→  totalDiscount: number;
    39→  paymentMethod: PaymentMethod;
    40→  setPaymentMethod: (m: PaymentMethod) => void;
    41→  cashReceived: string;
    42→  setCashReceived: (v: string) => void;
    43→  change: number;
    44→  selectedCustomer: string;
    45→  customers: CustomerItem[];
    46→  splitCashAmount: string;
    47→  setSplitCashAmount: (v: string) => void;
    48→  splitMpesaAmount: string;
    49→  setSplitMpesaAmount: (v: string) => void;
    50→  mpesaPhone: string;
    51→  setMpesaPhone: (v: string) => void;
    52→  mpesaStatus: 'idle' | 'processing' | 'success' | 'failed';
    53→  setMpesaStatus: (s: 'idle' | 'processing' | 'success' | 'failed') => void;
    54→  stkCheckoutRequestId: string;
    55→  stkResultDesc: string;

    57→  mpesaMutation: { isPending: boolean; mutate: (data: { phoneNumber: string; amount: number; accountReference: string; transactionDesc: string }) => void };
    58→  checkoutMutation: { isPending: boolean };
    59→  onSendStkPush: () => void;
    60→  onRetryStk: () => void;
    61→  onCompleteSale: () => void;
    62→  /** Optional cart items for the review/confirm step order summary */
    63→  cartItems?: Array<{
    64→    productId: string;
    65→    productName: string;
    66→    quantity: number;
    67→    pricePerUnit: number;
    68→    lineTotal: number;
    69→    discountPercent?: number;
    70→  }>;
    71→  /** Subtotal (pre-discount, pre-tax) for the order summary */
    72→  subtotal?: number;
    73→  /** Tax amount for the order summary */
    74→  taxAmount?: number;
    75→}
    76→

    80→  icon: React.ElementType;
    81→  desc: string;
    82→  color: string;
    83→  /** Tailwind gradient classes for the selected state border + icon tint */
    84→  selectedGradient: string;
    85→  /** Tailwind gradient for the icon circle when selected */
    86→  iconGradient: string;
    87→}[] = [
    88→  {
    89→    value: 'CASH', label: 'Cash', icon: Banknote, desc: 'Receive cash & give change', color: 'text-green-600',
    90→    selectedGradient: 'from-emerald-500 to-green-600',
    91→    iconGradient: 'bg-gradient-to-br from-emerald-500 to-green-600 text-white',
    92→  },
    93→  {
    94→    value: 'DEBT', label: 'Debt', icon: Wallet, desc: 'Put it on a customer account', color: 'text-amber-600',
    95→    selectedGradient: 'from-amber-500 to-orange-600',
    96→    iconGradient: 'bg-gradient-to-br from-amber-500 to-orange-600 text-white',
    97→  },
    98→  {
    99→    value: 'SPLIT', label: 'Either / Split', icon: Split, desc: 'Cash + M-Pesa combined', color: 'text-purple-600',
   100→    selectedGradient: 'from-fuchsia-500 to-purple-600',
   101→    iconGradient: 'bg-gradient-to-br from-fuchsia-500 to-purple-600 text-white',
   102→  },
   103→  {
   104→    value: 'MPESA', label: 'M-Pesa', icon: Smartphone, desc: 'STK push to customer phone', color: 'text-blue-600',
   105→    selectedGradient: 'from-sky-500 to-blue-600',
   106→    iconGradient: 'bg-gradient-to-br from-sky-500 to-blue-600 text-white',
   107→  },
   108→];
   109→
   110→type Step = 1 | 2 | 3;
   111→
   112→export function CheckoutDialog(props: CheckoutDialogProps) {
   113→  const {
   114→    open, onOpenChange, finalTotal, totalDiscount,
   115→    paymentMethod, setPaymentMethod,
   116→    cashReceived, setCashReceived, change,
   117→    selectedCustomer, customers,
   118→    splitCashAmount, setSplitCashAmount, splitMpesaAmount, setSplitMpesaAmount,
   119→    mpesaPhone, setMpesaPhone, mpesaStatus, setMpesaStatus,
   120→    stkCheckoutRequestId, stkResultDesc, stkPolling,
   121→    mpesaMutation, checkoutMutation,
   122→    onSendStkPush, onRetryStk, onCompleteSale,
   123→    cartItems, subtotal, taxAmount,
   124→  } = props;
   125→
   126→  const customer = customers.find((c) => c.id === selectedCustomer);
   127→  const debtAvailable = customer ? Math.max(0, customer.debtLimit - customer.currentDebtBalance) : 0;
   128→  const exceedsDebt = customer ? finalTotal > debtAvailable : false;
   129→
   130→  const cashValid = !!cashReceived && Number(cashReceived) >= finalTotal;
   131→  const splitValid = (Number(splitCashAmount) || 0) + (Number(splitMpesaAmount) || 0) >= finalTotal && (Number(splitMpesaAmount) || 0) > 0 ? !!mpesaPhone && mpesaPhone.length >= 9 : true;
   132→  const mpesaValid = !!mpesaPhone && mpesaPhone.length >= 9;
   133→
   134→  // Step state — start at step 1 (Customer & Payment Method)
   135→  const [step, setStep] = useState<Step>(1);
   136→  // Direction of step transition (for slide animation)
   137→  const [stepDirection, setStepDirection] = useState<'right' | 'left'>('right');
   138→
   139→  // Respect prefers-reduced-motion: when true, framer-motion transitions are
   140→  // reduced to opacity-only (no translate/scale) for vestibular safety.
   141→  const prefersReducedMotion = useReducedMotion();
   142→
   143→  // Reset M-Pesa status when payment method changes (away from MPESA)
   144→  useEffect(() => {
   145→    if (paymentMethod !== 'MPESA' && paymentMethod !== 'SPLIT' && mpesaStatus !== 'idle') {
   146→      setMpesaStatus('idle');
   147→    }
   148→  }, [paymentMethod, mpesaStatus, setMpesaStatus]);
   149→
   150→  // Reset to step 1 when dialog opens
   151→  // (deferred via requestAnimationFrame to avoid cascading-renders lint warning)
   152→  useEffect(() => {
   153→    if (!open) return;
   154→    const id = requestAnimationFrame(() => {
   155→      setStep(1);
   156→      setStepDirection('right');
   157→    });
   158→    return () => cancelAnimationFrame(id);
   159→  }, [open]);
   160→
   161→  const isProcessingMpesa = mpesaStatus === 'processing' || mpesaStatus === 'success' || mpesaStatus === 'failed';
   162→  const showMpesaPanel = (paymentMethod === 'MPESA' || paymentMethod === 'SPLIT');
   163→
   164→  // Compute derived totals for the order summary
   165→  // FINANCIAL MATH AUDIT: VAT is INSIDE the (VAT-inclusive) line totals.
   166→  // The fallback tax preview extracts the per-line VAT component in Decimal
   167→  // (never `Math.round(subtotal × 0.16)` on top — that double-counted VAT
   168→  // and diverged from the server's per-line taxRate computation).
   169→  const summarySubtotal = subtotal ?? cartItems?.reduce((sum, i) => sum + i.lineTotal, 0) ?? 0;
   170→  const summaryTax = taxAmount ?? round2(
   171→    (cartItems ?? []).reduce((sum, i) => {
   172→      const rate = Math.min(100, Math.max(0, i.taxRate || 0));
   173→      if (rate === 0) return sum;
   174→      const gross = toDec(i.lineTotal);
   175→      return sum.plus(gross.minus(gross.div(1 + rate / 100).toDecimalPlaces(2, Decimal.ROUND_HALF_UP)));
   176→    }, new Decimal(0))
   177→  );
   178→  const summaryTotalItems = cartItems?.reduce((sum, i) => sum + i.quantity, 0) ?? 0;
   179→
   180→  // Validation per step
   181→  const step1Valid = true; // Payment method is always selected (defaults to CASH)
   182→  const step2Valid = useMemo(() => {
   183→    if (paymentMethod === 'CASH') return cashValid;
   184→    if (paymentMethod === 'DEBT') return !!selectedCustomer && selectedCustomer !== 'walk-in' && !exceedsDebt;
   185→    if (paymentMethod === 'SPLIT') return splitValid;
   186→    if (paymentMethod === 'MPESA') return mpesaValid;
   187→    return false;
   188→  }, [paymentMethod, cashValid, selectedCustomer, exceedsDebt, splitValid, mpesaValid]);
   189→
   190→  // Show the M-Pesa processing panel as a special "step 2.5" — when mpesa is processing/success/failed,
   191→  // we still show step 2 content (STK status) but the footer changes via renderFooter.
   192→  const goToStep = (nextStep: Step) => {
   193→    if (nextStep > step) setStepDirection('right');
   194→    else setStepDirection('left');
   195→    setStep(nextStep);
   196→  };
   197→
   198→  // Build the footer buttons depending on state
   199→  const renderFooter = () => {
   200→    // M-Pesa processing/success/failed states override step navigation
   201→    if (mpesaStatus === 'processing') {
   202→      return (
   203→        <>
   204→          <Button variant="outline" onClick={() => { onRetryStk(); }} disabled={stkPolling}>
   205→            Cancel
   206→          </Button>
   207→          <a
   208→            href="https://daraja.safaricom.co.ke"
   209→            target="_blank"
   210→            rel="noopener noreferrer"
   211→            className="inline-flex items-center gap-1.5 h-9 px-3 rounded-md border border-input bg-background text-sm font-medium hover:bg-muted"
   212→          >
   213→            <ExternalLink className="h-3.5 w-3.5" /> Open M-Pesa Daraja
   214→          </a>
   215→        </>
   216→      );
   217→    }
   218→    if (mpesaStatus === 'success') {
   219→      return (
   220→        <>
   221→          <Button variant="outline" onClick={() => onOpenChange(false)}>Close</Button>
   222→          <Button
   223→            onClick={onCompleteSale}
   224→            disabled={checkoutMutation.isPending}
   225→            className="bg-green-600 hover:bg-green-700 text-white btn-press"
   226→          >
   227→            {checkoutMutation.isPending ? (
   228→              <><Loader2 className="mr-2 h-4 w-4 animate-spin" />Completing…</>
   229→            ) : (
   230→              <><CheckCircle className="mr-2 h-4 w-4" />Complete Sale</>
   231→            )}
   232→          </Button>
   233→        </>
   234→      );
   235→    }
   236→    if (mpesaStatus === 'failed') {
   237→      return (
   238→        <>
   239→          <Button variant="outline" onClick={() => onOpenChange(false)}>Cancel</Button>
   240→          <Button onClick={onRetryStk}>Try Again</Button>
   241→        </>
   242→      );
   243→    }
   244→
   245→    // Step-aware footer: Back/Next/Process Payment
   246→    if (step === 1) {
   247→      return (
   248→        <>
   249→          <Button variant="outline" onClick={() => onOpenChange(false)}>Cancel</Button>
   250→          <Button
   251→            onClick={() => goToStep(2)}
   252→            disabled={!step1Valid}
   253→            className="btn-press"
   254→          >
   255→            Next: Payment Details
   256→            <ChevronRight className="ml-1.5 h-4 w-4" />
   257→          </Button>
   258→        </>
   259→      );
   260→    }
   261→    if (step === 2) {
   262→      // For M-Pesa: show "Send STK Push" instead of "Next"
   263→      if (showMpesaPanel && !isProcessingMpesa) {
   264→        return (
   265→          <>
   266→            <Button variant="outline" onClick={() => goToStep(1)}>
   267→              <ChevronLeft className="mr-1.5 h-4 w-4" />
   268→              Back
   269→            </Button>
   270→            <Button
   271→              onClick={onSendStkPush}
   272→              disabled={mpesaMutation.isPending || !mpesaValid}
   273→              className="bg-green-600 hover:bg-green-700 text-white btn-press"
   274→            >
   275→              {mpesaMutation.isPending ? (
   276→                <><Loader2 className="mr-2 h-4 w-4 animate-spin" />Sending STK…</>
   277→              ) : (
   278→                <><Smartphone className="mr-2 h-4 w-4" />Send STK Push</>
   279→              )}
   280→            </Button>
   281→          </>
   282→        );
   283→      }
   284→      return (
   285→        <>
   286→          <Button variant="outline" onClick={() => goToStep(1)}>
   287→            <ChevronLeft className="mr-1.5 h-4 w-4" />
   288→            Back
   289→          </Button>
   290→          <Button
   291→            onClick={() => goToStep(3)}
   292→            disabled={!step2Valid}
   293→            className="btn-press"
   294→          >
   295→            Next: Review
   296→            <ChevronRight className="ml-1.5 h-4 w-4" />
   297→          </Button>
   298→        </>
   299→      );
   300→    }
   301→    // Step 3: Review & Confirm — "Process Payment" with gradient + loading spinner
   302→    return (
   303→      <>
   304→        <Button variant="outline" onClick={() => goToStep(2)} disabled={isProcessingMpesa}>
   305→          <ChevronLeft className="mr-1.5 h-4 w-4" />
   306→          Back
   307→        </Button>
   308→        <Button
   309→          onClick={onCompleteSale}
   310→          disabled={checkoutMutation.isPending || !step2Valid}
   311→          className="relative overflow-hidden bg-gradient-to-r from-emerald-600 via-emerald-500 to-amber-500 hover:from-emerald-700 hover:via-emerald-600 hover:to-amber-600 text-white shadow-lg shadow-emerald-500/30 btn-press"
   312→        >
   313→          {/* Glossy gradient sheen on top */}
   314→          <span className="pointer-events-none absolute inset-x-0 top-0 h-1/2 bg-gradient-to-b from-white/20 to-transparent" aria-hidden />
   315→          {checkoutMutation.isPending ? (
   316→            <span className="relative z-10 flex items-center">
   317→              <Loader2 className="mr-2 h-4 w-4 animate-spin" />Processing…
   318→            </span>
   319→          ) : (
   320→            <span className="relative z-10 flex items-center">
   321→              <CheckCircle className="mr-2 h-4 w-4" />Process Payment
   322→            </span>
   323→          )}
   324→        </Button>
   325→      </>
   326→    );
   327→  };
   328→
   329→  // Step indicator (3 steps with checkmarks for completed steps)
   330→  const renderStepIndicator = () => {
   331→    const steps = [
   332→      { n: 1 as Step, label: 'Customer & Method', icon: User },
   333→      { n: 2 as Step, label: 'Payment Details', icon: CreditCard },
   334→      { n: 3 as Step, label: 'Review & Confirm', icon: Receipt },
   335→    ];
   336→    return (
   337→      <div className="flex items-center justify-between gap-2 mb-4 select-none">
   338→        {steps.map((s, idx) => {
   339→          const StepIcon = s.icon;
   340→          const isComplete = step > s.n;
   341→          const isCurrent = step === s.n;
   342→          return (
   343→            <React.Fragment key={s.n}>
   344→              <div className="flex flex-col items-center gap-1 flex-1 min-w-0">
   345→                <div
   346→                  className={`relative flex items-center justify-center h-8 w-8 rounded-full border-2 transition-all ${
   347→                    isComplete
   348→                      ? 'bg-green-600 border-green-600 text-white'
   349→                      : isCurrent
   350→                        ? 'bg-primary border-primary text-primary-foreground animate-pulse-active'
   351→                        : 'bg-muted border-border text-muted-foreground'
   352→                  }`}
   353→                  aria-current={isCurrent ? 'step' : undefined}
   354→                  aria-label={`Step ${s.n}: ${s.label}${isComplete ? ' (completed)' : isCurrent ? ' (current)' : ''}`}
   355→                >
   356→                  {isComplete ? (
   357→                    <Check className="h-4 w-4" />
   358→                  ) : (
   359→                    <StepIcon className="h-3.5 w-3.5" />
   360→                  )}
   361→                </div>
   362→                <span className={`text-[9px] font-medium text-center leading-tight truncate w-full ${isCurrent ? 'text-foreground' : 'text-muted-foreground'}`}>
   363→                  {s.label}
   364→                </span>
   365→              </div>
   366→              {idx < steps.length - 1 && (
   367→                <div
   368→                  className={`h-0.5 flex-1 mt-[-12px] rounded transition-colors ${step > s.n ? 'bg-green-600' : 'bg-border'}`}
   369→                  aria-hidden
   370→                />
   371→              )}
   372→            </React.Fragment>
   373→          );
   374→        })}
   375→      </div>
   376→    );
   377→  };
   378→
   379→  // Order summary sidebar (used in steps 2 & 3)
   380→  const renderOrderSummary = (compact: boolean = false) => {
   381→    if (!cartItems || cartItems.length === 0) return null;
   382→    return (
   383→      <div className={`rounded-lg border bg-muted/30 p-3 space-y-2 ${compact ? 'text-xs' : 'text-sm'}`}>
   384→        <div className="flex items-center justify-between">
   385→          <span className="font-medium flex items-center gap-1.5">
   386→            <ShoppingCart className="h-3.5 w-3.5" />
   387→            Order Summary
   388→          </span>
   389→          <Badge variant="secondary" className="text-[10px]">
   390→            {summaryTotalItems} item{summaryTotalItems !== 1 ? 's' : ''}
   391→          </Badge>
   392→        </div>
   393→        <div className={`max-h-32 overflow-y-auto scrollbar-thin space-y-1 ${compact ? 'text-[11px]' : ''}`}>
   394→          {cartItems.map((item) => (
   395→            <div key={item.productId} className="flex items-center justify-between gap-2 py-0.5">
   396→              <div className="min-w-0 flex-1">
   397→                <p className="truncate">{item.productName}</p>
   398→                <p className="text-[10px] text-muted-foreground">
   399→                  {item.quantity} × {formatKES(item.pricePerUnit)}
   400→                  {item.discountPercent && item.discountPercent > 0 ? ` · -${item.discountPercent}%` : ''}
   401→                </p>
   402→              </div>
   403→              <span className="font-medium shrink-0">{formatKES(item.lineTotal)}</span>
   404→            </div>
   405→          ))}
   406→        </div>
   407→        <Separator />
   408→        <div className="space-y-0.5 text-[11px]">
   409→          <div className="flex justify-between">
   410→            <span className="text-muted-foreground">Subtotal (VAT incl.)</span>
   411→            <span>{formatKES(summarySubtotal)}</span>
   412→          </div>
   413→          {totalDiscount > 0 && (
   414→            <div className="flex justify-between text-green-600">
   415→              <span className="flex items-center gap-0.5"><Tag className="h-2.5 w-2.5" />Discount</span>
   416→              <span>-{formatKES(totalDiscount)}</span>
   417→            </div>
   418→          )}
   419→          <div className="flex justify-between">
   420→            <span className="text-muted-foreground">VAT (incl. in prices)</span>
   421→            <span>{formatKES(summaryTax)}</span>
   422→          </div>
   423→          <Separator />
   424→          <div className="flex justify-between font-bold">
   425→            <span>Total</span>
   426→            <span className="text-gradient">{formatKES(finalTotal)}</span>
   427→          </div>
   428→        </div>
   429→      </div>
   430→    );
   431→  };
   432→
   433→  // Success screen — shown briefly when mpesaStatus === 'success'
   434→  const renderSuccessScreen = () => (
   435→    <div className="text-center py-6 space-y-3 animate-scale-in">
   436→      <div className="relative mx-auto w-16 h-16">
   437→        <div className="absolute inset-0 rounded-full bg-green-100 dark:bg-green-950/40 animate-ping opacity-75" />
   438→        <div className="relative flex items-center justify-center w-16 h-16 rounded-full bg-green-600 text-white shadow-lg">
   439→          <Check className="h-8 w-8" />
   440→        </div>
   441→      </div>
   442→      <div>
   443→        <p className="text-lg font-bold text-green-700 dark:text-green-400">Payment Authorised!</p>
   444→        <p className="text-sm text-muted-foreground">
   445→          M-Pesa STK push was accepted. Tap &ldquo;Complete Sale&rdquo; to finalise the receipt.
   446→        </p>
   447→      </div>
   448→    </div>
   449→  );
   450→
   451→  // Framer-motion variants for step slide transitions (respects prefers-reduced-motion)
   452→  const stepVariants = {
   453→    enter: (dir: 'right' | 'left') => ({
   454→      opacity: 0,
   455→      x: prefersReducedMotion ? 0 : dir === 'right' ? 24 : -24,
   456→    }),
   457→    center: {
   458→      opacity: 1,
   459→      x: 0,
   460→    },
   461→    exit: (dir: 'right' | 'left') => ({
   462→      opacity: 0,
   463→      x: prefersReducedMotion ? 0 : dir === 'right' ? -24 : 24,
   464→    }),
   465→  };
   466→
   467→  const stepTransition = {
   468→    duration: prefersReducedMotion ? 0.15 : 0.3,
   469→    ease: [0.16, 1, 0.3, 1] as [number, number, number, number],
   470→  };
   471→
   472→  return (
   473→    <ResponsiveDialog
   474→      open={open}
   475→      onOpenChange={(o) => {
   476→        // Prevent closing while STK push is actively processing
   477→        if (!o && mpesaStatus === 'processing') return;
   478→        onOpenChange(o);
   479→      }}
   480→      title={
   481→        <span className="flex items-center gap-2">
   482→          <CreditCard className="h-5 w-5 text-primary" />
   483→          Complete Payment
   484→          {step > 1 && (
   485→            <Badge variant="outline" className="text-[10px] ml-1">Step {step}/3</Badge>
   486→          )}
   487→        </span>
   488→      }
   489→      description={
   490→        <span className="break-words">
   491→          Total: <span className="font-bold text-primary">{formatKES(finalTotal)}</span>
   492→          {totalDiscount > 0 && (
   493→            <span className="text-green-600 text-xs ml-2">(Save {formatKES(totalDiscount)})</span>
   494→          )}
   495→        </span>
   496→      }
   497→      size="md"
   498→      footer={renderFooter()}
   499→    >
   500→      <div className="space-y-4">
   501→        {/* Step progress indicator — always visible */}
   502→        {mpesaStatus !== 'success' && renderStepIndicator()}
