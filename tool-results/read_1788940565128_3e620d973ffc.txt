   150→   150→  return new Date(date).toLocaleDateString('en-GB', {

   152→   152→    month: 'short',

   157→   157→function daysSince(date: string | Date): number {
   158→   158→  const d = new Date(date);
   159→   159→  const now = new Date();
   160→   160→  return Math.floor((now.getTime() - d.getTime()) / (1000 * 60 * 60 * 24));
   161→   161→}

   378→   378→
   379→   379→function PipelineSection({ storeId }: { storeId: string }) {
   380→   380→  const queryClient = useQueryClient();
   381→   381→  const authUser = useAuthStore((s) => s.user);
   382→   382→  const canRun = authUser ? SENIOR_ROLES.includes(authUser.role) : false;
   383→   383→
   384→   384→  const [showScheduleResult, setShowScheduleResult] = useState(false);
   385→   385→  const [scheduleResult, setScheduleResult] = useState<ReminderScheduleResult | null>(null);
   386→   386→
   387→   387→  // Live reminder status counts.
   388→   388→  const { data: remindersData, isLoading } = useQuery({
   389→   389→    queryKey: ['debt-reminders-summary', storeId],
   390→   390→    queryFn: () => debtRemindersApi.list({ storeId, limit: 1 }),
   391→   391→    enabled: !!storeId,
   392→   392→    refetchInterval: 15_000, // Auto-refresh every 15s while the pipeline runs
   393→   393→  });
   394→   394→
   395→   395→  const summary = remindersData?.summary;
   396→   396→  const byStatus = summary?.byStatus ?? {};
   397→   397→  const byChannel = summary?.byChannel ?? {};
   398→   398→
   399→   399→  const scheduleMutation = useMutation({
   400→   400→    mutationFn: () => debtRemindersApi.schedule(storeId),
   401→   401→    onSuccess: (data) => {
   402→   402→      const result = data.data;
   403→   403→      setScheduleResult(result);
   404→   404→      setShowScheduleResult(true);
   405→   405→      toast.success(data.message || `Scheduled ${result.scheduled} reminder(s).`);
   406→   406→      queryClient.invalidateQueries({ queryKey: ['debt-reminders-summary', storeId] });
   407→   407→      queryClient.invalidateQueries({ queryKey: ['debt-overdue', storeId] });
   408→   408→      queryClient.invalidateQueries({ queryKey: ['debt-reminders', storeId] });
   409→   409→    },
   410→   410→    onError: (err) => handleError(err),
   411→   411→  });
   412→   412→
   413→   413→  const processMutation = useMutation({
   414→   414→    mutationFn: () => debtRemindersApi.process(storeId),
   415→   415→    onSuccess: (data) => {
   416→   416→      const result = data.data;
   417→   417→      toast.success(data.message || `Processed ${result.total} reminders: ${result.sent} sent, ${result.failed} failed.`);
   418→   418→      queryClient.invalidateQueries({ queryKey: ['debt-reminders-summary', storeId] });
   419→   419→      queryClient.invalidateQueries({ queryKey: ['debt-reminders', storeId] });
   420→   420→    },
   421→   421→    onError: (err) => handleError(err),
   422→   422→  });
   423→   423→
   424→   424→  const pendingCount = byStatus.PENDING ?? 0;
   425→   425→  const sentCount = byStatus.SENT ?? 0;
   426→   426→  const deliveredCount = byStatus.DELIVERED ?? 0;
   427→   427→  const failedCount = byStatus.FAILED ?? 0;
   428→   428→  const totalReminders = summary?.total ?? 0;
   429→   429→
   430→   430→  const successRate = sentCount + deliveredCount + failedCount > 0
   431→   431→    ? Math.round(((sentCount + deliveredCount) / (sentCount + deliveredCount + failedCount)) * 100)
   432→   432→    : 0;
   433→   433→
   434→   434→  if (isLoading) {
   435→   435→    return (
