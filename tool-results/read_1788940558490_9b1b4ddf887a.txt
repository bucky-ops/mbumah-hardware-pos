   149→function formatDate(date: string | Date): string {
   150→  return new Date(date).toLocaleDateString('en-GB', {
   151→    day: '2-digit',
   152→    month: 'short',
   153→    year: 'numeric',
   154→  });
   155→}
   156→
   157→function daysSince(date: string | Date): number {
   158→  const d = new Date(date);
   159→  const now = new Date();
   160→  return Math.floor((now.getTime() - d.getTime()) / (1000 * 60 * 60 * 24));
   161→}

   379→function PipelineSection({ storeId }: { storeId: string }) {
   380→  const queryClient = useQueryClient();
   381→  const authUser = useAuthStore((s) => s.user);
   382→  const canRun = authUser ? SENIOR_ROLES.includes(authUser.role) : false;
   383→
   384→  const [showScheduleResult, setShowScheduleResult] = useState(false);
   385→  const [scheduleResult, setScheduleResult] = useState<ReminderScheduleResult | null>(null);
   386→
   387→  // Live reminder status counts.
   388→  const { data: remindersData, isLoading } = useQuery({
   389→    queryKey: ['debt-reminders-summary', storeId],
   390→    queryFn: () => debtRemindersApi.list({ storeId, limit: 1 }),
   391→    enabled: !!storeId,
   392→    refetchInterval: 15_000, // Auto-refresh every 15s while the pipeline runs
   393→  });
   394→
   395→  const summary = remindersData?.summary;
   396→  const byStatus = summary?.byStatus ?? {};
   397→  const byChannel = summary?.byChannel ?? {};
   398→
   399→  const scheduleMutation = useMutation({
   400→    mutationFn: () => debtRemindersApi.schedule(storeId),
   401→    onSuccess: (data) => {
   402→      const result = data.data;
   403→      setScheduleResult(result);
   404→      setShowScheduleResult(true);
   405→      toast.success(data.message || `Scheduled ${result.scheduled} reminder(s).`);
   406→      queryClient.invalidateQueries({ queryKey: ['debt-reminders-summary', storeId] });
   407→      queryClient.invalidateQueries({ queryKey: ['debt-overdue', storeId] });
   408→      queryClient.invalidateQueries({ queryKey: ['debt-reminders', storeId] });
   409→    },
   410→    onError: (err) => handleError(err),
   411→  });
   412→
   413→  const processMutation = useMutation({
   414→    mutationFn: () => debtRemindersApi.process(storeId),
   415→    onSuccess: (data) => {
   416→      const result = data.data;
   417→      toast.success(data.message || `Processed ${result.total} reminders: ${result.sent} sent, ${result.failed} failed.`);
   418→      queryClient.invalidateQueries({ queryKey: ['debt-reminders-summary', storeId] });
   419→      queryClient.invalidateQueries({ queryKey: ['debt-reminders', storeId] });
   420→    },
   421→    onError: (err) => handleError(err),
   422→  });
   423→
   424→  const pendingCount = byStatus.PENDING ?? 0;
   425→  const sentCount = byStatus.SENT ?? 0;
   426→  const deliveredCount = byStatus.DELIVERED ?? 0;
   427→  const failedCount = byStatus.FAILED ?? 0;
   428→  const totalReminders = summary?.total ?? 0;
   429→
   430→  const successRate = sentCount + deliveredCount + failedCount > 0
   431→    ? Math.round(((sentCount + deliveredCount) / (sentCount + deliveredCount + failedCount)) * 100)
   432→    : 0;
   433→
   434→  if (isLoading) {
   435→    return (
   436→      <div className="space-y-4">
   437→        <Skeleton className="h-48 w-full" />
   438→        <Skeleton className="h-64 w-full" />
   439→      </div>
   440→    );
   441→  }
   442→
   443→  return (
   444→    <div className="space-y-4">
   445→      {/* Pipeline cards */}
   446→      <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
   447→        {/* Schedule card */}
   448→        <Card className="border-blue-200 dark:border-blue-900/40">
   449→          <CardHeader>
   450→            <CardTitle className="text-base flex items-center gap-2">
   451→              <div className="p-1.5 rounded-lg bg-blue-100 dark:bg-blue-900/30">
   452→                <Zap className="h-4 w-4 text-blue-600 dark:text-blue-400" />
   453→              </div>
   454→              Step 1: Schedule Reminders
   455→            </CardTitle>
   456→            <CardDescription>
   457→              Scan overdue debts and create PENDING reminder tasks based on aging escalation rules.
   458→              Fast — no network calls.
   459→            </CardDescription>
   460→          </CardHeader>
   461→          <CardContent className="space-y-3">
   462→            <div className="text-xs text-muted-foreground space-y-1">
   463→              <p>• <span className="font-medium">1–30 days late</span>: weekly (SMS + WhatsApp)</p>
   464→              <p>• <span className="font-medium">31–60 days late</span>: every 3 days (SMS + WhatsApp + Email)</p>
   465→              <p>• <span className="font-medium">61+ days late</span>: daily + manager escalation</p>
   466→            </div>
   467→            <Button
   468→              className="w-full"
   469→              onClick={() => scheduleMutation.mutate()}
   470→              disabled={!canRun || scheduleMutation.isPending}
   471→            >
   472→              {scheduleMutation.isPending ? (
   473→                <>
   474→                  <Loader2 className="h-4 w-4 mr-1.5 animate-spin" />
   475→                  Scheduling...
   476→                </>
   477→              ) : (
   478→                <>
   479→                  <Zap className="h-4 w-4 mr-1.5" />
   480→                  Schedule Reminders Now
   481→                </>
   482→              )}
   483→            </Button>
   484→            {!canRun && (
   485→              <p className="text-xs text-amber-600 dark:text-amber-400 text-center">
   486→                Manager+ role required to run the scheduler.
   487→              </p>
   488→            )}
   489→          </CardContent>
   490→        </Card>
   491→
   492→        {/* Process card */}
   493→        <Card className="border-emerald-200 dark:border-emerald-900/40">
   494→          <CardHeader>
   495→            <CardTitle className="text-base flex items-center gap-2">
   496→              <div className="p-1.5 rounded-lg bg-emerald-100 dark:bg-emerald-900/30">
   497→                <Send className="h-4 w-4 text-emerald-600 dark:text-emerald-400" />
   498→              </div>
   499→              Step 2: Process Pending
   500→            </CardTitle>
   501→            <CardDescription>
   502→              Send all PENDING reminders via SMS (Twilio), WhatsApp (Twilio), and Email (Resend).
   503→              Network-bound — batches of 100.
   504→            </CardDescription>
   505→          </CardHeader>
   506→          <CardContent className="space-y-3">
   507→            <div className="flex items-center justify-between p-3 rounded-lg bg-muted/50">
   508→              <div>
   509→                <p className="text-xs text-muted-foreground">Pending in queue</p>
   510→                <p className="text-2xl font-bold text-blue-600 dark:text-blue-400">{pendingCount}</p>
   511→              </div>
   512→              <Activity className="h-8 w-8 text-blue-600/40 dark:text-blue-400/40" />
   513→            </div>
   514→            <Button
   515→              className="w-full"
   516→              variant="default"
   517→              onClick={() => processMutation.mutate()}
   518→              disabled={!canRun || processMutation.isPending || pendingCount === 0}
   519→            >
   520→              {processMutation.isPending ? (

   533→              <p className="text-xs text-muted-foreground text-center">

   535→              </p>
   536→            )}
   537→          </CardContent>
   538→        </Card>
   539→      </div>
   540→
   541→      {/* Status overview */}
   542→      <Card>
   543→        <CardHeader className="pb-3">
   544→          <CardTitle className="text-base flex items-center gap-2">
   545→            <Activity className="h-4 w-4" />

   552→          <div className="grid grid-cols-2 md:grid-cols-4 gap-3">
   553→            <StatusTile label="Pending" value={pendingCount} icon={Clock} color="text-slate-600 dark:text-slate-300" bg="bg-slate-100 dark:bg-slate-900/40" />

   586→            </div>

   590→            <div className="text-center py-6 text-sm text-muted-foreground">

   595→      </Card>
   596→

   601→            <DialogTitle className="flex items-center gap-2">

   603→              Scheduling Complete
   604→            </DialogTitle>

   611→              <div className="grid grid-cols-3 gap-3">
   612→                <div className="p-3 rounded-lg bg-emerald-50 dark:bg-emerald-950/20 border border-emerald-200 dark:border-emerald-900/40 text-center">
   613→                  <p className="text-2xl font-bold text-emerald-600 dark:text-emerald-400">{scheduleResult.scheduled}</p>

   615→                </div>
   616→                <div className="p-3 rounded-lg bg-slate-50 dark:bg-slate-900/20 border text-center">
   617→                  <p className="text-2xl font-bold text-slate-600 dark:text-slate-400">{scheduleResult.skipped}</p>
   618→                  <p className="text-xs text-muted-foreground">Skipped</p>
   619→                </div>
   620→                <div className="p-3 rounded-lg bg-rose-50 dark:bg-rose-950/20 border border-rose-200 dark:border-rose-900/40 text-center">
   621→                  <p className="text-2xl font-bold text-rose-600 dark:text-rose-400">{scheduleResult.errors}</p>
   622→                  <p className="text-xs text-muted-foreground">Errors</p>
   623→                </div>
   624→              </div>
   625→              {scheduleResult.details.length > 0 && (
   626→                <ScrollArea className="max-h-64">
   627→                  <div className="space-y-1">
   628→                    {scheduleResult.details.map((d, i) => (
   629→                      <div key={i} className="flex items-center gap-2 text-xs p-2 rounded border">
   630→                        {d.scheduled ? (
   631→                          <CheckCircle className="h-3.5 w-3.5 text-emerald-600 shrink-0" />
   632→                        ) : (
   633→                          <Clock className="h-3.5 w-3.5 text-slate-400 shrink-0" />
   634→                        )}
   635→                        <span className="font-mono text-muted-foreground">{d.debtLedgerId}</span>
   636→                        <span className="text-muted-foreground truncate flex-1">
   637→                          {d.reason || 'Scheduled'}
   638→                        </span>
   639→                      </div>
   640→                    ))}
   641→                  </div>
   642→                </ScrollArea>
   643→              )}
   644→              <p className="text-xs text-muted-foreground">
   645→                Run <span className="font-medium">Step 2: Process Pending</span> to actually send the queued reminders.
   646→              </p>
   647→            </div>
   648→          )}
   649→          <DialogFooter>
   650→            <Button onClick={() => setShowScheduleResult(false)}>Close</Button>
   651→          </DialogFooter>
   652→        </DialogContent>
   653→      </Dialog>
   654→    </div>
   655→  );
