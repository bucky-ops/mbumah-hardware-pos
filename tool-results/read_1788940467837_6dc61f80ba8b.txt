    57→function getAvatarGradient(name: string): string {
    58→  let hash = 0;
    59→  for (let i = 0; i < name.length; i++) {
    60→    hash = name.charCodeAt(i) + ((hash << 5) - hash);
    61→  }
    62→  return AVATAR_GRADIENTS[Math.abs(hash) % AVATAR_GRADIENTS.length];
    63→}

   106→// Auto-format phone number to Kenyan format
   107→function formatKenyanPhone(value: string): string {
   108→  const digits = value.replace(/\D/g, '');
   109→  if (digits.startsWith('254')) {
   110→    return `+${digits.slice(0, 3)} ${digits.slice(3, 6)} ${digits.slice(6, 9)}`;
   111→  }
   112→  if (digits.startsWith('0') && digits.length >= 4) {
   113→    return `${digits.slice(0, 4)} ${digits.slice(4, 7)} ${digits.slice(7, 10)}`.trim();
   114→  }
   115→  return value;
   116→}

   147→/**
   148→ * Map an API timeline entry (uppercase type, `timestamp`/`ref` fields) into
   149→ * the display shape this dialog expects (lowercase type, `date`/`reference`).
   150→ * The API returns `CustomerHistoryTimelineEntry` which has an index signature
   151→ * so we can read amount/discountAmount/status/etc. defensively.
   152→ */
   153→function normalizeHistoryEntry(raw: CustomerHistoryTimelineEntry): HistoryEntry {
   154→  const typeMap: Record<string, HistoryEntry['type']> = {
   155→    SALE: 'sale',
   156→    INVOICE: 'invoice',
   157→    CREDIT: 'credit',
   158→    GIFT_CARD_REDEMPTION: 'gift_card',
   159→    VOUCHER_REDEMPTION: 'voucher',
   160→    DEBT_PAYMENT: 'payment',
   161→    DELIVERY_NOTE: 'delivery_note',
   162→  };
   163→  const type = typeMap[raw.type] ?? 'sale';
   164→  const reference = String(raw.ref ?? raw.id ?? '');
   165→  const date = raw.timestamp;
   166→  // VOUCHER_REDEMPTION uses `discountAmount` instead of `amount`.
   167→  const rawAmount =
   168→    raw.type === 'VOUCHER_REDEMPTION'
   169→      ? (raw.discountAmount as unknown as number | undefined)
   170→      : (raw.amount as unknown as number | undefined);
   171→  const amount = typeof rawAmount === 'number' && !Number.isNaN(rawAmount) ? rawAmount : 0;
   172→  const status =
   173→    (raw.status as unknown as string | undefined) ??
   174→    (raw.paymentStatus as unknown as string | undefined) ??
   175→    (raw.invoiceType as unknown as string | undefined) ??
   176→    (raw.creditType as unknown as string | undefined);
   177→  const description =
   178→    (raw.description as unknown as string | null | undefined) ??
   179→    (raw.deliveryAddress as unknown as string | null | undefined) ??
   180→    undefined;
   181→  return { id: raw.id, date, type, reference, amount, status, description };
   182→}

   189→    case 'gift_card': return { label: 'Gift Card', icon: Gift, color: 'text-pink-600 dark:text-pink-400', bg: 'bg-pink-100 dark:bg-pink-900/30' };

   197→function CustomerHistoryDialog({
   198→  customer,
   199→  open,
   200→  onOpenChange,
   201→  storeId,
   202→}: {
   203→  customer: CustomerItem | null;
   204→  open: boolean;
   205→  onOpenChange: (open: boolean) => void;
   206→  storeId: string;
   207→}) {
   208→  const [typeFilter, setTypeFilter] = useState<string>('all');
   209→  const [sendingStatement, setSendingStatement] = useState(false);
   210→
   211→  const { data, isLoading, isError } = useQuery({
   212→    queryKey: ['customer-history', customer?.id, storeId],
   213→    queryFn: async (): Promise<CustomerHistoryData | null> => {
   214→      if (!customer) return null;
   215→      // Preferred path — typed API client (returns ApiResponse<CustomerHistoryResult>).
   216→      try {
   217→        const response = await customersApi.getHistory(customer.id);
   218→        const payload: CustomerHistoryResult | undefined = response?.data;
   219→        if (payload) {
   220→          return {
   221→            customer: payload.customer,
   222→            summary: payload.summary,
   223→            timeline: Array.isArray(payload.timeline)
   224→              ? payload.timeline.map(normalizeHistoryEntry)
   225→              : [],
   226→          };
   227→        }
   228→      } catch {
   229→        // fall through to direct fetch
   230→      }
   231→      // Fallback: direct fetch with same-origin credentials.
   232→      const res = await fetch(`/api/customers/${customer.id}/history`, {
   233→        credentials: 'same-origin',
   234→      });
   235→      const json = (await res.json()) as {
   236→        success?: boolean;
   237→        data?: CustomerHistoryResult;
   238→        error?: string;
   239→      };
   240→      if (!json.success || !json.data) {
   241→        throw new Error(json.error || 'Failed to load history');
   242→      }
   243→      const payload = json.data;
   244→      return {
   245→        customer: payload.customer,
   246→        summary: payload.summary,
   247→        timeline: Array.isArray(payload.timeline)
   248→          ? payload.timeline.map(normalizeHistoryEntry)
   249→          : [],
   250→      };
   251→    },
   252→    enabled: !!customer && open,
   253→  });
   254→
   255→  const summary = data?.summary;
   256→  const timeline: HistoryEntry[] = data?.timeline ?? [];
   257→  const filteredTimeline = useMemo(() => {
   258→    if (typeFilter === 'all') return timeline;
   259→    return timeline.filter((e) => e.type === typeFilter);
   260→  }, [timeline, typeFilter]);
   261→
   262→  async function handleSendStatement() {
   263→    if (!customer) return;
   264→    const phone = prompt('Enter WhatsApp phone number to send the statement to:', customer.phone || '') || '';
   265→    if (!phone) return;
   266→    setSendingStatement(true);
   267→    try {
   268→      const res = await whatsappApi.sendDocument({
   269→        type: 'statement',
   270→        documentId: customer.id,
   271→        customerId: customer.id,
   272→        storeId,
   273→        phone,
   274→      });
   275→      // whatsappApi.sendDocument returns ApiResponse<{waLink,...}> — unwrap .data
   276→      const waLink = res?.data?.waLink;
   277→      const title = res?.data?.documentTitle;
   278→      if (waLink) {
   279→        window.open(waLink, '_blank');
   280→        toast.success(`${title ?? 'Statement'} sent via WhatsApp`);
   281→      } else {
   282→        toast.error('No WhatsApp link returned by the server.');
   283→      }
   284→    } catch (err) {
   285→      const msg = handleError(err, 'Send statement via WhatsApp');
   286→      toast.error(msg);
   287→    } finally {
   288→      setSendingStatement(false);
   289→    }
   290→  }
   291→
   292→  if (!customer) return null;
   293→  const loyalty = getLoyaltyTier(customer.loyaltyPoints);
   294→  const gradient = getAvatarGradient(customer.name);
   295→
   296→  return (
   297→    <ResponsiveDialog
   298→      open={open}
   299→      onOpenChange={onOpenChange}
   300→      size="2xl"
   301→      title={
   302→        <span className="flex items-center gap-2">
   303→          <History className="h-5 w-5 text-primary" />
   304→          Account History
   305→        </span>
   306→      }
   307→      description={`Full statement & activity timeline for ${customer.name}`}
   308→      footer={
   309→        <>
   310→          <Button variant="outline" onClick={() => onOpenChange(false)}>Close</Button>
   311→          <Button
   312→            onClick={handleSendStatement}
   313→            disabled={sendingStatement}
   314→            className="bg-green-600 hover:bg-green-700 text-white gap-2"
   315→          >
   316→            {sendingStatement ? <Loader2 className="h-4 w-4 animate-spin" /> : <Send className="h-4 w-4" />}
   317→            Send Statement via WhatsApp
   318→          </Button>
   319→        </>
   320→      }
   321→    >
   322→      <div className="space-y-5">
   323→        {/* Customer Summary Card */}
   324→        <Card className="border-l-4 border-l-primary">
   325→          <CardContent className="p-4 space-y-4">
   326→            <div className="flex items-start gap-3">

   329→                  {customer.name.split(' ').map(n => n[0]).join('').slice(0, 2)}

   336→                    {loyalty.icon} {loyalty.tier} Member
   337→                  </span>
   338→                  <DebtStatusBadge customer={customer} />
